# по умолчанию присваивается str
a = input()
print(a)       # 123 → '123'

# оборачивание int() если нужно числовое значение
a = int(input()) 
print(a)    # 123 → 123

 
# множественное присваивание str с пробелом в качестве разделителя
#(в .split() можно указать любой разделитель,
#отсутствие аргумента в .split() означает использование пробела в качестве разделителя)
a, b, c = input().split() 
print(b)       # scissors paper stone → paper

# множественное присваивание int с пробелом в качестве разделителя
#(в .split() можно указать любой разделитель,
#отсутствие аргумента в .split() означает использование пробела в качестве разделителя)
a, b, c = map(int, input().split()) 
print(b)   # 1 2 3 → 2

# создание списка из каждого символа введенной строки
#(пробел при вводе тоже воспринимается как символ)
a = list(input()) 
print(a)         # letters → ['l', 'e', 't', 't', 'e', 'r', 's']

# создание списка из каждой цифры введенного числа
#(пробелы при вводе вызовут ошибку)
a = list(map(int, input())) 
print(a)       # 123456 → [1, 2, 3, 4, 5, 6]

 
# создание списка из отдельных строк
#(в .split() можно указать любой разделитель,
#отсутствие аргумента в .split() означает использование пробела в качестве разделителя)
a = list(input().split()) 
print(a)       # letters words → ['letters', 'words']

# тоже самое что и a = list(input().split())
a = input().split() 
print(a)    # letters words → ['letters', 'words']

# создание списка из отдельных чисел
#(в .split() можно указать любой разделитель,
#отсутствие аргумента в .split() означает использование пробела в качестве разделителя)
a = list(map(int, input().split())) 
print(a)           # 123 456 → [123, 456]
